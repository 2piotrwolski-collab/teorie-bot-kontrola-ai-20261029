"""One fixed GC session: free-cloud tick/quote/OI availability, NO orders/PnL."""
from AlgorithmImports import *
from datetime import datetime
import json,hashlib

class TurtleTickAvailability(QCAlgorithm):
    def initialize(self):
        if self.live_mode:raise ValueError("Historical only")
        self.set_time_zone(TimeZones.NEW_YORK)
        self.set_start_date(2024,1,16);self.set_end_date(2024,1,16)
        self.set_cash(500000)
        self.report={"protocol":"T003_GC_TICK_OI_AVAILABILITY_20261030_V1",
            "orders_allowed":False,"strategy_return_calculated":False,
            "scope":"Jan2 full source session only; not full study or execution validation",
            "request_start_NY":"2024-01-01 18:00:00","request_end_NY":"2024-01-02 17:00:00",
            "contracts":[],"errors":[]}
        for expiry in [datetime(2024,2,27),datetime(2024,4,26)]:
            symbol=Symbol.create_future("GC",Market.COMEX,expiry)
            self.add_future_contract(symbol,Resolution.MINUTE,fill_forward=False,extended_market_hours=True)
            row={"sid":str(symbol.id),"symbol":str(symbol),"ticks":0,"trade_ticks":0,"quote_ticks":0,
                "invalid_two_sided_quotes":0,"zero_or_negative_quote_size":0,
                "unordered_ticks":0,"same_time_adjacent_ticks":0,"trades_without_valid_prior_quote":0,
                "trades_strict_prior_quote_age_le_1s":0,"trades_strict_prior_quote_age_le_5s":0,
                "trades_latest_quote_same_timestamp":0,"quote_older_than_5s":0}
            previous=None;quote=None
            try:
                ticks=self.history[Tick](symbol,datetime(2024,1,1,18),datetime(2024,1,2,17),Resolution.TICK,
                    fill_forward=False,extended_market_hours=True)
                for tick in ticks:
                    row["ticks"]+=1
                    if previous is not None:
                        row["unordered_ticks"]+=int(tick.time<previous)
                        row["same_time_adjacent_ticks"]+=int(tick.time==previous)
                    previous=tick.time
                    if row["ticks"]==1:row["first_tick"]=str(tick.time)
                    row["last_tick"]=str(tick.time)
                    if tick.tick_type==TickType.QUOTE:
                        row["quote_ticks"]+=1
                        valid=tick.bid_price>0 and tick.ask_price>=tick.bid_price
                        row["invalid_two_sided_quotes"]+=int(not valid)
                        row["zero_or_negative_quote_size"]+=int(tick.bid_size<=0 or tick.ask_size<=0)
                        # Invalid latest update invalidates previous quote; never carry it as fresh.
                        quote=(tick.time,valid and tick.bid_size>0 and tick.ask_size>0)
                    elif tick.tick_type==TickType.TRADE:
                        row["trade_ticks"]+=1
                        if quote is None or not quote[1]:row["trades_without_valid_prior_quote"]+=1
                        else:
                            age=(tick.time-quote[0]).total_seconds()
                            if age==0:row["trades_latest_quote_same_timestamp"]+=1
                            elif age>0:
                                row["trades_strict_prior_quote_age_le_1s"]+=int(age<=1)
                                row["trades_strict_prior_quote_age_le_5s"]+=int(age<=5)
                                row["quote_older_than_5s"]+=int(age>5)
            except Exception as exc:row["tick_error"]={"type":type(exc).__name__,"message":str(exc)}
            try:
                oi=list(self.history[OpenInterest](symbol,datetime(2024,1,2),datetime(2024,1,13),Resolution.MINUTE,
                    fill_forward=False,extended_market_hours=True))
                positives=[b for b in oi if b.value>0]
                row["open_interest"]={"count":len(oi),"positive_count":len(positives),
                    "first_positive_history_timestamp":str(positives[0].end_time) if positives else None,
                    "first_positive_value":float(positives[0].value) if positives else None,
                    "last_history_timestamp":str(oi[-1].end_time) if oi else None,
                    "publication_latency_asof_verified":False}
            except Exception as exc:row["open_interest_error"]={"type":type(exc).__name__,"message":str(exc)}
            row["historical_timestamp_is_not_real_receipt_time"]=True
            row["full_execution_gate_passed"]=False
            self.report["contracts"].append(row)

    def on_data(self,data):pass

    def on_end_of_algorithm(self):
        self.report["status"]="AVAILABILITY_AUDIT_ONLY_NOT_STRATEGY_RESULT"
        raw=json.dumps(self.report,sort_keys=True,separators=(",",":"))
        chunks=[raw[i:i+4500] for i in range(0,len(raw),4500)]
        for i,s in enumerate(chunks):self.debug("T003_TICK_CHUNK "+json.dumps({"i":i,"n":len(chunks),"data":s},separators=(",",":")))
        self.debug("T003_TICK_COMPLETE "+json.dumps({"chunks":len(chunks),"bytes":len(raw.encode()),"sha256":hashlib.sha256(raw.encode()).hexdigest()}))
