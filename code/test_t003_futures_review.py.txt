import json,unittest,copy
from pathlib import Path
from t003_futures_review import merge
ROOT=Path(__file__).parent

class FuturesProjectionTests(unittest.TestCase):
    def test_fullrange_data_and_margin_clock_is_not_strategy_outcome(self):
        before=json.loads((ROOT/'research_round15/T003_GC_2019_2025/current_view_before.json').read_text())
        review=json.loads((ROOT/'results/t003_futures_review_20261030_v4.json').read_text())
        after=merge(before,review)
        self.assertEqual(merge(after,review),after)
        for a,b in zip(before['items'],after['items']):
            if a['id']!='T003':self.assertEqual(a,b)
        card=next(c for c in after['items'] if c['id']=='T003')
        self.assertEqual(card['dividend_reconciliation_status'],'closed_numerical_reconstruction_discrepancy')
        self.assertIn('4 635 018',card['observations'][0]['summary'])
        self.assertIn('margin',card['futures_original_test_status'].lower())
        self.assertEqual(len(card['findings']),len(set(card['findings'])))
        self.assertTrue(any('705,32' in s for s in card['findings']))
        self.assertTrue(all(g['status']=='partial' for g in card['gap_updates']))
    def test_actual_roll_view_preserves_old_scope_and_other_cards(self):
        before=json.loads((ROOT/'results/research_progress_round4.json').read_text())
        review=json.loads((ROOT/'results/t003_futures_review_20261030_v3.json').read_text())
        after=merge(before,review)
        self.assertEqual(merge(after,review),after)
        for a,b in zip(before['items'],after['items']):
            if a['id']!='T003':self.assertEqual(a,b)
        card=next(c for c in after['items'] if c['id']=='T003')
        self.assertEqual(card['dividend_reconciliation_status'],'closed_numerical_reconstruction_discrepancy')
        self.assertIn('705,32',card['observations'][0]['summary'])
        self.assertEqual(len(card['findings']),len(set(card['findings'])))
        self.assertTrue(all(g['status']=='partial' for g in card['gap_updates']))
    def test_new_account_view_is_idempotent_without_reopening_dividends(self):
        before=json.loads((ROOT/'results/research_progress_round4.json').read_text())
        review=json.loads((ROOT/'results/t003_futures_review_20261030_v2.json').read_text())
        after=merge(before,review)
        self.assertEqual(merge(after,review),after)
        for a,b in zip(before['items'],after['items']):
            if a['id']!='T003':self.assertEqual(a,b)
        card=next(c for c in after['items'] if c['id']=='T003')
        self.assertEqual(card['dividend_reconciliation_status'],'closed_numerical_reconstruction_discrepancy')
        self.assertIn('56974',card['observations'][0]['summary'])
        self.assertEqual(len(card['findings']),len(set(card['findings'])))
    def test_scoped_current_view_and_idempotence(self):
        before=json.loads((ROOT/'results/research_progress_round4.json').read_text())
        review=json.loads((ROOT/'results/t003_futures_review_20261030_v1.json').read_text())
        original=copy.deepcopy(before);after=merge(before,review)
        self.assertEqual(before,original)
        self.assertEqual(merge(after,review),after)
        for a,b in zip(before['items'],after['items']):
            if a['id']!='T003':self.assertEqual(a,b)
        a=next(c for c in before['items'] if c['id']=='T003');b=next(c for c in after['items'] if c['id']=='T003')
        self.assertEqual(a['dividend_reconciliation_status'],b['dividend_reconciliation_status'])
        self.assertEqual(a['execution_evidence_file'],b['execution_evidence_file'])
        self.assertEqual([g['gap_id'] for g in a['gap_updates']],[g['gap_id'] for g in b['gap_updates']])
        self.assertEqual(a['overall_status'],b['overall_status'])
        self.assertTrue(all(g['status']=='partial' for g in b['gap_updates']))

if __name__=='__main__':unittest.main()
