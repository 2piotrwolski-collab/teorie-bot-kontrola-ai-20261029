"""Maintain a scoped dividend-accounting reconciliation on the current T003 card."""
import copy
import hashlib
import json
from pathlib import Path
from data_io import atomic_json

    card=next(x for x in out['items'] if x['id']=='T003')
    card['t003_dividend_audit_sha256']=review['audit_sha256']
    card['findings']=[s for s in card['findings'] if not s.startswith(('B-exact na rzeczywistych zamrożonych dywidendach cash','Uzgodnienie dywidend 30.10:'))]
    card['findings'] += [
        'Uzgodnienie dywidend 30.10: niezależny kod odtworzył rekonstrukcję Claude na zamrożonym SPY. W latach 2020–27.10.2026 reinwestycja daje CAGR 2,800930%, gotówka 2,785713%, dotychczasowy matched SPY 15,127449%, przy 5 bps na stronę. Sygnały i 40 operacji są identyczne; wynik nie rozstrzyga oryginalnego Turtle.',
        'Uzgodnienie dywidend 30.10: reinwestowanie po zamknięciu ex-date zamiast trzymania dywidendy w gotówce wyjaśnia 99,39% różnicy do raportowanych 2,801%. Pozostałe składniki: koszt także od dywidendy przy sprzedaży na ex-date, rok 365,25 zamiast 365,2425 i zaokrąglenie wyniku. Nie zmieniano reguł ani progów strategii.',
        'Uzgodnienie dywidend 30.10: Claude nie zachował pierwotnego CSV ani pełnego kodu. Otrzymano opis nowej rekonstrukcji, bez wymienionej w nim paczki JS/CSV. Wyjaśnienie rachunkowe zostało sprawdzone; zgodność z utraconym pierwotnym przebiegiem i jego danymi pozostaje nieudowodniona.',
        'Uzgodnienie dywidend 30.10: reinwestycja bez kosztu na zamknięciu ex-date i natychmiastowy kredyt cash są założeniami audytu. Nie potwierdzono dat wypłat, realnego DRIP, podatków, FX ani filli; naliczanie prowizji od dywidendy odtworzono wyłącznie jako diagnostykę różnicy.'
    ]
    for observation in card.get('observations',[]):
        if observation['id'].startswith('T003_AB_SENSITIVITY_'):
            observation['limitations']=[
                'B-exact zachowuje dywidendy cash do kolejnego wejścia. Różnicę rachunkową do nowej rekonstrukcji Claude wyjaśniono w audycie z 30.10; pierwotne pliki Claude nie są zachowane.'
                if s=='B-exact zachowuje dywidendy cash do kolejnego wejścia, bez natychmiastowej reinwestycji. Różnica wobec Claude pozostaje jawna.' else s
                for s in observation.get('limitations',[])]
    card['observations']=[o for o in card.get('observations',[]) if not o['id'].startswith('T003_DIVIDEND_RECON_')]
    labels={'development':'2001–2015','validation':'2016–2019','retrospective_check':'2020–27.10.2026'}
    rows=[[labels[x['segment']],round(x['cash_CAGR_pct'],6),round(x['reconstructed_Claude_CAGR_pct'],6),round(x['reconstruction_minus_cash_pp'],7),round(x['matched_archival_SPY_CAGR_pct'],6)] for x in audit['summary']]
    card['observations'].insert(0,{'id':'T003_DIVIDEND_RECON_'+review['audit_sha256'][:12],
        'title':'Uzgodnienie dywidend z rekonstrukcją Claude · 30.10.2026',
        'summary':'CAGR % rocznie. Te same zamrożone dane i operacje; odmienna reinwestycja i konwencje rachunkowe. Nie jest to nowy test oryginalnego Turtle.',
        'columns':['Okres','Dywidendy w gotówce','Rekonstrukcja Claude','Różnica pp','Dotychczasowy SPY'],
        'rows':rows,'limitations':[
            'Potwierdzono rekonstrukcję na danych Codex; nie udowodniono utraconego pierwotnego CSV/kodu Claude ani identyczności jego dawnego pobrania Yahoo.',
            'Nie otrzymano paczki JS/CSV wymienionej w opisie Claude. Nowe obliczenia wykonano własnym kodem na podstawie czterech opisanych różnic.',
            'Natychmiastowy kredyt na ex-date, darmowa reinwestycja po close i koszt od dywidendy to model księgowy, bez potwierdzenia rzeczywistych wypłat, DRIP, podatków, FX i filli.',
            'Stare wyniki i ich hashe zachowane. Pozostałe braki pełnego Turtle na kontraktach nadal otwarte.'],
        'evidence_report':review['public_report_url']})
    card['observation_ids']=[o['id'] for o in card['observations']]
    card['sensitivity_audit_next_action']=review['next_action']
    card['dividend_audit_next_action']=review['next_action']
    return out


def refresh(root, db):
    root=Path(root)
    pointer=root/'config/t003_dividend_current.json'
    if not pointer.exists():return
    p=json.loads(pointer.read_text())
    review_file=root/p['review_file']
    if hashlib.sha256(review_file.read_bytes()).hexdigest()!=p['review_sha256']:
        raise ValueError('Dividend review identity mismatch')
    review=json.loads(review_file.read_text())
    audit_file=root/review['audit_file']
    if hashlib.sha256(audit_file.read_bytes()).hexdigest()!=review['audit_sha256']:
        raise ValueError('Dividend audit identity mismatch')
    audit=json.loads(audit_file.read_text())
    path=root/'results/research_progress_round4.json'
    old=json.loads(path.read_text());current=merge(old,review,audit)
    if current!=old:
        atomic_json(path,current)
        db.research_update('T003',next(x for x in current['items'] if x['id']=='T003'),p['review_file'])
    atomic_json(root/'reports/AUDYT_T003_DYWIDENDY.json',audit)
