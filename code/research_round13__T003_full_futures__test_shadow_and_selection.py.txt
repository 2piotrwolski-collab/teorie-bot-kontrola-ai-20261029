import unittest,copy
from datetime import datetime,timedelta,timezone
from account_engine import Account,Contract,Margin,Quote,D,DataGate
from system2 import System2
from shadow_system2 import ShadowSystem2
T=datetime(2024,1,2,15,tzinfo=timezone.utc)
def levels(c,**kw):return {'contract':c,'available_at':T,'buy':105,'short':95,'exit_long':90,'exit_short':110,'n':2,**kw}
def quote(c,t,bid=106,ask=106.1,size=10):return Quote(c,t,bid,ask,t,t,size,size)
class IntegrationTests(unittest.TestCase):
 def book(self,capital=100000):
  cs=[Contract('GC1','GC',100,'.1'),Contract('GC2','GC',100,'.1')]
  a=Account(capital,cs,{'GC':'2.51'},[],[],synthetic=True)
  for c in cs:a.set_margin(c.symbol,Margin(1000,900,T,T+timedelta(days=100),'fixture','synthetic_fixture'),T)
  s=System2(a)
  for c in cs:s.set_levels(levels(c.symbol))
  return a,s
 def test_strongest_eligible_month_selected_before_any_fill(self):
  a,s=self.book();s.select_many('GC',[('GC1',100,T),('GC2',90,T)],T)
  s.on_trades(T+timedelta(seconds=1),{'GC1':106,'GC2':106})
  self.assertEqual([u.contract for u in a.units.values()],['GC2'])
 def test_weakest_short_month_selected(self):
  a,s=self.book();s.select_many('GC',[('GC1',100,T),('GC2',110,T)],T)
  s.on_trades(T+timedelta(seconds=1),{'GC1':94,'GC2':94})
  self.assertEqual([u.contract for u in a.units.values()],['GC2'])
 def test_nonbreakout_month_not_selected_despite_strength(self):
  a,s=self.book();s.select_many('GC',[('GC1',100,T),('GC2',0,T)],T)
  s.on_trades(T+timedelta(seconds=1),{'GC1':106,'GC2':104})
  self.assertEqual([u.contract for u in a.units.values()],['GC1'])
 def test_equal_time_marks_do_not_invent_notional_cut(self):
  cs=[Contract('A','A',1,'.1'),Contract('B','B',1,'.1')]
  a=Account(10000,cs,{'A':0,'B':0},[],[],slippage_ticks=0,synthetic=True)
  for c in cs:
   a.set_margin(c.symbol,Margin(1,1,T,T+timedelta(days=5),'fixture','synthetic_fixture'),T);a.mark(c.symbol,100,T)
   a.reserve_unit(c.symbol,c.symbol,1,1,100,T)
  for c in cs:
   a.fill_unit(c.symbol,quote(c.symbol,T+timedelta(seconds=1),100,100,100))
  a.mark_batch({'A':90,'B':105},T+timedelta(seconds=2))
  self.assertEqual(a.equity(),D(9500));self.assertEqual(a.notional.amount,D(10000))
 def test_incomplete_settlement_cannot_partially_mutate_book(self):
  cs=[Contract('A','A',1,'.1'),Contract('B','B',1,'.1')]
  a=Account(10000,cs,{'A':0,'B':0},[],[],synthetic=True)
  for c in cs:
   a.set_margin(c.symbol,Margin(1,1,T,T+timedelta(days=5),'fixture','synthetic_fixture'),T);a.mark(c.symbol,100,T)
   a.reserve_unit(c.symbol,c.symbol,1,1,1,T)
  for c in cs:a.fill_unit(c.symbol,quote(c.symbol,T+timedelta(seconds=1),100,100))
  before=copy.deepcopy(a.units)
  with self.assertRaises(DataGate):a.settle({'A':200},T+timedelta(seconds=2),'fixture')
  self.assertEqual(a.units,before)
 def test_missing_fee_cannot_silently_become_zero(self):
  with self.assertRaises(DataGate):Account(10000,[Contract('A','A',1,'.1')],{},[],[],synthetic=True)
class ShadowTests(unittest.TestCase):
 def setUp(self):self.s=ShadowSystem2('GC2','.1');self.s.set_levels(levels('GC2'))
 def trade(self,p,seconds):self.s.on_trade(p,T+timedelta(seconds=seconds))
 def q(self,b,a,seconds,size=10):return self.s.on_quote(quote('GC2',T+timedelta(seconds=seconds),b,a,size))
 def result(self,seconds):
  t=T+timedelta(seconds=seconds);self.s.certify_coverage(t,'fixture complete tick replay');return self.s.roll_eligibility(t)
 def test_no_inherited_old_contract_position(self):
  self.trade(100,1);self.assertEqual(self.result(2)['direction'],0)
 def test_own_trigger_alone_is_not_filled_position(self):
  self.trade(106,1);self.assertEqual(self.result(2)['direction'],0)
 def test_own_later_quote_qualifies_successor(self):
  self.trade(106,1);self.q(106,106.1,2);self.assertEqual(self.result(3)['direction'],1)
 def test_own_stop_removes_roll_eligibility_before_quote(self):
  self.trade(106,1);self.q(106,106.1,2);self.trade(100,3)
  self.assertEqual(self.result(4)['direction'],0)
 def test_no_covering_future_replay_after_old_watermark(self):
  self.trade(100,1);self.result(2)
  with self.assertRaises(DataGate):self.trade(200,2)
 def test_coverage_timestamp_alone_not_accepted(self):
  self.trade(100,1)
  with self.assertRaises(DataGate):self.s.roll_eligibility(T+timedelta(seconds=2))
 def test_gap_fill_actual_and_source_stops(self):
  self.trade(120,1);self.q(121,121.1,2)
  self.assertEqual(self.s.units[0]['entry'],D('121.2'));self.assertEqual(self.s.units[0]['stop'],D('117.2'))
 def test_own_pyramid_advances_older_stops_halfN(self):
  self.trade(106,1);self.q(106,106.1,2);self.trade(108,3);self.q(108,108.1,4)
  self.assertEqual(len(self.s.units),2);self.assertEqual(self.s.units[0]['stop'],D('103.2'))
 def test_own_four_unit_limit(self):
  for i,p in enumerate([106,110,114,118,122]):self.trade(p,i*2+1);self.q(p,p+.1,i*2+2)
  self.assertEqual(len(self.s.units),4)
 def test_same_time_quote_cannot_fill_own_signal(self):
  self.trade(106,1);self.assertFalse(self.q(106,106.1,1));self.assertEqual(self.s.direction,0)
 def test_wrong_raw_contract_rejected(self):
  with self.assertRaises(DataGate):self.s.on_quote(quote('GC1',T+timedelta(seconds=1)))
if __name__=='__main__':unittest.main()
