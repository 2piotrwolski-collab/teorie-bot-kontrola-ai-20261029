"""Boundary controls for the distinct knowledge/effectivity clocks."""
import unittest
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo
from dated_margin import DatedRate, MarginTimeline, MarginEvidenceError

NY = ZoneInfo('America/New_York')


class DatedMarginTests(unittest.TestCase):
    def setUp(self):
        self.old = DatedRate('GC_TEST', '10065', '9150',
            datetime(2020, 6, 17, 23, 59, 59, tzinfo=NY),
            datetime(2020, 6, 18, 17, tzinfo=NY), 'Synthetic prior rate',
            'notice_date_end_of_day_convention')
        self.new = DatedRate('GC_TEST', '9020', '8200',
            datetime(2020, 6, 18, 23, 59, 59, tzinfo=NY),
            datetime(2020, 6, 19, 17, tzinfo=NY),
            'CME20-247;17NYclose is explicit standard-session interpretation',
            'notice_date_end_of_day_convention')
        self.start = datetime(2020, 6, 19, tzinfo=NY)
        self.end = datetime(2020, 6, 22, 17, tzinfo=NY)
        self.timeline = MarginTimeline('GC_TEST', [self.old, self.new],
            self.start, self.end, allow_notice_date_convention=True)

    def test_known_future_rate_not_used_at_midnight_or_during_day(self):
        for hour in (0, 9, 16):
            self.assertEqual(self.timeline.lookup(self.start.replace(hour=hour)), self.old)

    def test_after_close_is_strict_and_updates_on_next_actual_session(self):
        self.assertEqual(self.timeline.lookup(self.new.effective_after), self.old)
        self.assertEqual(self.timeline.lookup(self.new.effective_after + timedelta(microseconds=1)), self.new)
        self.assertEqual(self.timeline.lookup(self.start.replace(hour=18)), self.new)

    def test_UTC_matches_EDT(self):
        self.assertEqual(self.timeline.lookup(datetime(2020, 6, 19, 20, 59, tzinfo=timezone.utc)), self.old)
        self.assertEqual(self.timeline.lookup(datetime(2020, 6, 19, 21, 1, tzinfo=timezone.utc)), self.new)

    def test_EST_is_different_offset(self):
        self.assertEqual(datetime(2020, 1, 9, 17, tzinfo=NY).astimezone(timezone.utc).hour, 22)
        self.assertEqual(self.new.effective_after.astimezone(timezone.utc).hour, 21)

    def test_coverage_end_not_extrapolated(self):
        with self.assertRaises(MarginEvidenceError):
            self.timeline.lookup(self.end + timedelta(seconds=1))

    def test_no_first_future_row_fallback(self):
        t = MarginTimeline('GC_TEST', [self.new], self.start, self.end, True)
        with self.assertRaises(MarginEvidenceError):
            t.lookup(self.start)

    def test_no_unannounced_margin_or_naive_clock(self):
        with self.assertRaises(MarginEvidenceError):
            DatedRate('GC_TEST', 9020, 8200, self.new.effective_after + timedelta(seconds=1),
                self.new.effective_after, 'Notice', 'archived_publication_timestamp')
        with self.assertRaises(MarginEvidenceError):
            self.timeline.lookup(datetime(2020, 6, 19))

    def test_convention_is_not_silently_publication_evidence(self):
        strict = MarginTimeline('GC_TEST', [self.old, self.new], self.start, self.end)
        with self.assertRaises(MarginEvidenceError):
            strict.lookup(self.start.replace(hour=18))

    def test_future_extension_cannot_change_earlier_rate(self):
        future = DatedRate('GC_TEST', 11000, 10000, self.end,
            self.end + timedelta(days=1), 'Future fixture', 'archived_publication_timestamp')
        t = MarginTimeline('GC_TEST', [self.old, self.new, future], self.start, self.end, True)
        self.assertEqual(t.lookup(self.start.replace(hour=9)), self.old)

    def test_duplicate_effectivity_and_wrong_contract_rejected(self):
        with self.assertRaises(MarginEvidenceError):
            MarginTimeline('GC_TEST', [self.old, self.old], self.start, self.end, True)
        with self.assertRaises(MarginEvidenceError):
            MarginTimeline('OTHER', [self.old], self.start, self.end, True)


if __name__ == '__main__':
    unittest.main()
